CREATE DATABASE GAJOS;
USE GAJOS;

CREATE TABLE TIPO_DOCUMENTO (
ID_TIPO INT (10) NOT NULL PRIMARY KEY AUTO_INCREMENT,
TIPO_DOCUMENTO VARCHAR (50) NOT NULL,
ESTADO BOOLEAN NOT NULL
);


CREATE TABLE ROL (
ID_ROL INT (20) NOT NULL PRIMARY KEY AUTO_INCREMENT,
NOMBRE_ROL VARCHAR (50) NOT NULL,
DESCRIPCION TEXT NOT NULL,
ESTADO BOOLEAN NOT NULL
);

CREATE TABLE TRABAJADOR (
DOCUMENTO VARCHAR (10) NOT NULL PRIMARY KEY,
ID_TIPO INT (10) NOT NULL,
ID_ROL INT (20) NOT NULL,
N1 VARCHAR (100) NOT NULL,
N2 VARCHAR (100),
AP1 VARCHAR (100) NOT NULL,
AP2 VARCHAR (100) NOT NULL,
ESTADO BOOLEAN NOT NULL,
CONSTRAINT TRABAJADOR_TIPO FOREIGN KEY (ID_TIPO) REFERENCES TIPO_DOCUMENTO (ID_TIPO),
CONSTRAINT TRABAJOR_ROL FOREIGN KEY (ID_ROL) REFERENCES ROL (ID_ROL)
);


CREATE TABLE PERMISO (
ID_PERMISO INT (100) NOT NULL PRIMARY KEY AUTO_INCREMENT,
NOMBRE_PERMISO VARCHAR (100) NOT NULL,
ESTADO BOOLEAN NOT NULL,
DESCRIPCION TEXT NOT NULL
);


CREATE TABLE ROL_TIENE_PERMISO (
ID_ROL_PERMISO INT (100) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_ROL INT (20) NOT NULL,
ID_PERMISO INT (100) NOT NULL,
ESTADO BOOLEAN NOT NULL,
CONSTRAINT ROL_PERMISO_PERMISO FOREIGN KEY (ID_PERMISO) REFERENCES PERMISO (ID_PERMISO),
CONSTRAINT ROL_PERMISO_ROL FOREIGN KEY (ID_ROL) REFERENCES ROL (ID_ROL)
);


CREATE TABLE VEHICULO (
ID_PLACA VARCHAR (50) NOT NULL PRIMARY KEY,
MODELO VARCHAR (50) NOT NULL,
MARCA_CARRO VARCHAR (50) NOT NULL,
ESTADO BOOLEAN NOT NULL
);


CREATE TABLE SELECCION (
DOCUMENTO VARCHAR (10) NOT NULL,
ID_PLACA VARCHAR (50) NOT NULL,
KILOMETRAJE_INICIAL VARCHAR (50) NOT NULL,
KILOMETRAJE_FINAL VARCHAR (50) NOT NULL,
FECHA_INICIO VARCHAR (50),
FECHA_FINAL VARCHAR (50),
HORA_INICIO VARCHAR (50),
HORA_FINAL VARCHAR (50),
NOVEDADES TEXT NOT NULL,
CONSTRAINT SELECCION_VEHICULO FOREIGN KEY (ID_PLACA) REFERENCES VEHICULO (ID_PLACA), 
CONSTRAINT SELECCION_TRABAJADOR FOREIGN KEY (DOCUMENTO) REFERENCES TRABAJADOR (DOCUMENTO)
);




CREATE TABLE COMBUSTIBLE (
ID_COMBUSTIBLE INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ESTADO BOOLEAN NOT NULL,
NOMBRE_COMBUSTIBLE VARCHAR (50) NOT NULL
);

CREATE TABLE SUMINISTRAR (
ID_SUMINISTRO INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_PLACA VARCHAR (50) NOT NULL,
ID_COMBUSTIBLE INT (50) NOT NULL,
KILOMETRAJE VARCHAR (50) NOT NULL,
FECHA_SUMINISTRO VARCHAR (50) NOT NULL,
GALONES VARCHAR (50) NOT NULL,
HORA_SUMINISTRO VARCHAR (50) NOT NULL,
CONSTRAINT SUMINISTAR_VEHICULO FOREIGN KEY (ID_PLACA) REFERENCES VEHICULO (ID_PLACA),
CONSTRAINT SUMINISTRAR_COMBUSTIBLE FOREIGN KEY (ID_COMBUSTIBLE) REFERENCES COMBUSTIBLE (ID_COMBUSTIBLE)
);


CREATE TABLE CORRECCION (
ID_CORRECION INT(50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_PLACA VARCHAR(50) NOT NULL,
DESCRIPCION TEXT NOT NULL,
UBICACION VARCHAR (100) NOT NULL,
CONSTRAINT CORRECCION_PLACA FOREIGN KEY (ID_PLACA) REFERENCES VEHICULO(ID_PLACA)
);

CREATE TABLE LLANTA (
ID_LLANTA INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
TIPO_LLANTA VARCHAR (50) NOT NULL,
DESCRIPCION TEXT NOT NULL
);

CREATE TABLE VEHICULO_TIENE_LLANTA (
ID_VEHICULO_LLANTA INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_LLANTA INT(50) NOT NULL,
ID_PLACA VARCHAR (50) NOT NULL,
CONSTRAINT VEHICULO_TIENE_LLANTA_LLANTA FOREIGN KEY (ID_LLANTA) REFERENCES LLANTA (ID_LLANTA),
CONSTRAINT VEHICULO_TIENE_LLANTA_VEHICULO FOREIGN KEY (ID_PLACA) REFERENCES VEHICULO (ID_PLACA)
);



CREATE TABLE ACEITE (
ID_ACEITE INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
TIPO_ACEITE VARCHAR (50) NOT NULL,
FECHA VARCHAR (50) NOT NULL,
DESCRIPCION TEXT NOT NULL
);

CREATE TABLE VEHICULO_SUMINISTRO_ACEITE (
ID_VEHICULO_SUMINISTRO_ACEITE INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_ACEITE INT (50) NOT NULL,
ID_PLACA VARCHAR (50) NOT NULL,
FECHA VARCHAR (50) NOT NULL,
CONSTRAINT VEHICULO_SUMINISTRO_ACEITE FOREIGN KEY (ID_ACEITE) REFERENCES ACEITE (ID_ACEITE),
CONSTRAINT VEHICULO_SUMINISTRO_ACEITE_VEHICULO FOREIGN KEY (ID_PLACA) REFERENCES VEHICULO (ID_PLACA)
);



CREATE TABLE MANTENIMIENTO (
ID_MANTENIMIENTO INT (50) NOT NULL PRIMARY KEY AUTO_INCREMENT,
ID_PLACA INT (50) NOT NULL,
DESCRIPCION TEXT NOT NULL
);
