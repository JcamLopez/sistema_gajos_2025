
CREATE TABLE TRABAJADOR (
  DOCUMENTO VARCHAR(10) PRIMARY KEY NOT NULL,
  FK_ID_TIPO INT(50) NOT NULL,
  N1 VARCHAR(50) NOT NULL,
  N2 VARCHAR(50),
  AP1 VARCHAR(50) NOT NULL,
  AP2 VARCHAR(50),
  ESTADO BOOLEAN DEFAULT TRUE,
  FOREIGN KEY (FK_ID_TIPO) REFERENCES TIPO_DOCUMENTO(ID_TIPO)
);


CREATE TABLE TIPO_DOCUMENTO (
  ID_TIPO INT(50) PRIMARY KEY NOT NULL AUTO_INCREMENT,
  TIPO_DOCUMENTO VARCHAR(50) NOT NULL,
  ESTADO BOOLEAN DEFAULT TRUE,
);


CREATE TABLE VEHICULO(
  ID_PLACA INT(50) AUTO_INCREMENT NOT NULL,
  MODELO VARCHAR(50) NOT NULL,
  MARCA_CARRO VARCHAR(50) NOT NULL,
  ESTADO BOOLEAN NOT NULL 
);
